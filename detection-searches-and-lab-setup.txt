SECTION 3: SPL DETECTION SEARCHES (Splunk Enterprise 9.4.1)

# ATTACK SCENARIO 1: HYDRA BRUTE FORCE

# Search 1: Naive Detection (Benign Window)
# Returns all failed login events with no grouping or threshold
# Result: 41 events

index=main sourcetype=linux_secure "Failed password" earliest="07/28/2026:00:00:00" latest="08/03/2026:00:35:04"

---

# Search 2: V1 Grouped by Source IP (Benign Window)
# Groups failed logins by source IP and flags any IP with more than 5 failures
# Result: 1 blunt result showing shared IP 192.168.80.128

index=main sourcetype=linux_secure "Failed password" earliest="07/28/2026:00:00:00" latest="08/03/2026:00:35:04"
| rex field=_raw "from (?<attacker_ip>\d+\.\d+\.\d+\.\d+)"
| stats count by attacker_ip
| where count > 5

---

# Search 3: V2 Tuned by Target User (Benign Window)
# Groups failed logins by targeted user account and flags accounts with more than 5 failures
# Result: employee5 with 17 failures and employee1 with 6 failures

index=main sourcetype=linux_secure "Failed password" earliest="07/28/2026:00:00:00" latest="08/03/2026:00:35:04"
| rex field=_raw "Failed password for (invalid user )?(?<target_user>\w+) from"
| stats count by target_user
| where count > 5

---

# Search 4: Attack Window Detection Confirmation
# Same tuned logic applied to attack window to confirm genuine attack is still detectable
# Result: root with 310 failed logins

index=main sourcetype=linux_secure "Failed password" earliest="07/22/2026:09:41:00" latest="07/22/2026:09:52:00"
| rex field=_raw "Failed password for (invalid user )?(?<target_user>\w+) from"
| stats count by target_user
| where count > 5

---

# ATTACK SCENARIO 2: SUDO-BASED PRIVILEGE ESCALATION

# Search 5: Raw Broad Search
# Searches all sudo related events including machine generated noise
# Result: 32 events

index=main sourcetype=linux_secure sudo earliest="08/06/2026:14:14:00" latest="08/06/2026:14:32:00"
| stats count

---

# Search 6: Refined to Real Commands Only
# Filters to real user invoked sudo commands by requiring COMMAND field
# Result: 11 events showing dara 10 and employee1 1

index=main sourcetype=linux_secure process=sudo COMMAND=* earliest="08/06/2026:14:14:00" latest="08/06/2026:14:32:00"
| rex "sudo:\s+(?<invoking_user>\w+)\s+:"
| stats count by invoking_user

---

# Search 7: Tuned Rule
# Restricts to commands run as root and excludes authorised administrator dara
# Result: employee1 only with count 1

index=main sourcetype=linux_secure process=sudo COMMAND=* USER=root earliest="08/06/2026:14:14:00" latest="08/06/2026:14:32:00"
| rex "sudo:\s+(?<invoking_user>\w+)\s+:"
| stats count by invoking_user
| search NOT invoking_user=dara

# Search 10: Tuned Rule
# Changes unit of analysis to distinct destination ports per source
# Flags any source contacting more than 10 distinct ports
# Result: 1 identified scanning source which is Kali at 192.168.80.128

index=main sourcetype=ufw earliest="08/12/2026:17:23:00" latest="08/12/2026:18:06:00"
| stats dc(DPT) as distinct_ports by SRC
